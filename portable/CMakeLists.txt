cmake_minimum_required(VERSION 3.24)
project(PortableLauncher LANGUAGES CXX RC)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Windows x64 and MSVC are required")
endif()
if(NOT EXISTS "${PORTABLE_RESOURCE_FILE}")
    message(FATAL_ERROR "Run build-portable.ps1 to prepare the embedded payload")
endif()
add_executable(TeamSpeakAudioControl-portable WIN32 launcher.cpp "${PORTABLE_RESOURCE_FILE}")
target_compile_features(TeamSpeakAudioControl-portable PRIVATE cxx_std_17)
target_compile_definitions(TeamSpeakAudioControl-portable PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00)
target_compile_options(TeamSpeakAudioControl-portable PRIVATE /utf-8 /W4)
set_target_properties(TeamSpeakAudioControl-portable PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_link_libraries(TeamSpeakAudioControl-portable PRIVATE bcrypt setupapi shell32 ole32)
